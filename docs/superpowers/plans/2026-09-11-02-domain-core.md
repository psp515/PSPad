# Domain Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The pure functional core — areas, lists, goals, tasks, steps and the Inbox — as decide/apply functions with no infrastructure, fully covered by tests.

**Architecture:** Each aggregate is a state record plus two static functions: `Decide(state, command) -> events` validates and returns what should happen; `Apply(state, event) -> state` is total and never throws. The current instant and any new ids arrive on the command, stamped by the caller, so the functions stay deterministic and compile to WebAssembly (spec §4, AD-3, AD-4).

**Tech Stack:** .NET 10, C# records, xUnit, Shouldly

**Spec:** `docs/superpowers/specs/2026-09-11-gtd-core-design.md`

## Global Constraints

- .NET 10 (LTS). All projects target `net10.0`.
- `PSPad.Domain` references no infrastructure packages and must compile for WebAssembly — the purity guard test from plan 01 Task 3 enforces this.
- All user-facing dates are `DateOnly`; all stored instants are UTC `DateTimeOffset`.
- Decide functions never read the clock, generate ids, or perform I/O.
- Code, comments, commits and docs in English.
- GPL v3 — every dependency must be license-compatible.

**Depends on:** plan 01 (solution scaffold, test projects).

```csharp
namespace PSPad.Domain.Common;

public static class Text
{
    public static string Required(string? value, int maxLength)
    {
        var trimmed = value?.Trim() ?? string.Empty;

        if (trimmed.Length == 0)
            throw new DomainException(ErrorCodes.NameRequired, "A name is required.");

        if (trimmed.Length > maxLength)
            throw new DomainException(
                ErrorCodes.NameTooLong, $"Maximum length is {maxLength} characters.");

        return trimmed;
    }

    public static string? Optional(string? value, int maxLength)
    {
        var trimmed = value?.Trim();

        if (string.IsNullOrEmpty(trimmed))
            return null;

        if (trimmed.Length > maxLength)
            throw new DomainException(
                ErrorCodes.TextTooLong, $"Maximum length is {maxLength} characters.");

        return trimmed;
    }
}
```

`src/PSPad.Domain/Areas/AreaDecider.cs`:

```csharp
using PSPad.Domain.Common;

namespace PSPad.Domain.Areas;

public static class AreaDecider
{
    const int MaxNameLength = 100;

    public static IReadOnlyList<IDomainEvent> Decide(AreaState? state, ICommand command) =>
        command switch
        {
            CreateArea create => Create(state, create),
            RenameArea rename => Rename(Require(state), rename),
            ArchiveArea archive => Archive(Require(state), archive),
            _ => []
        };

    static IReadOnlyList<IDomainEvent> Create(AreaState? state, CreateArea command)
    {
        if (state is not null)
            return [];

        var name = Text.Required(command.Name, MaxNameLength);

        return [new AreaCreated(
            command.UserId, command.IssuedAt, command.CommandId,
            command.AreaId, name, command.Order)];
    }

    static IReadOnlyList<IDomainEvent> Rename(AreaState state, RenameArea command)
    {
        RejectIfArchived(state);

        var name = Text.Required(command.Name, MaxNameLength);

        return name == state.Name
            ? []
            : [new AreaRenamed(command.UserId, command.IssuedAt, command.CommandId, state.Id, name)];
    }

    static IReadOnlyList<IDomainEvent> Archive(AreaState state, ArchiveArea command) =>
        state.Archived
            ? []
            : [new AreaArchived(command.UserId, command.IssuedAt, command.CommandId, state.Id)];

    static AreaState Require(AreaState? state) =>
        state ?? throw new DomainException(ErrorCodes.NotFound, "Area does not exist.");

    static void RejectIfArchived(AreaState state)
    {
        if (state.Archived)
            throw new DomainException(ErrorCodes.Archived, "Area is archived.");
    }

    public static AreaState? Apply(AreaState? state, IDomainEvent @event) =>
        (state, @event) switch
        {
            (_, AreaCreated e) => new AreaState(e.AreaId, e.UserId, e.Name, e.Order, Archived: false),
            ({ } s, AreaRenamed e) => s with { Name = e.Name },
            ({ } s, AreaArchived) => s with { Archived = true },
            _ => state
        };
}
```

```csharp
    const int MaxDescriptionLength = 2000;

    public static IReadOnlyList<IDomainEvent> Decide(GoalState? state, ICommand command) =>
        command switch
        {
            CreateGoal create => Create(state, create),
            EditGoal edit => Edit(Require(state), edit),
            ArchiveGoal archive => Archive(Require(state), archive),
            _ => []
        };

    {
        if (state is not null)

        var name = Text.Required(command.Name, MaxNameLength);
        var description = Text.Optional(command.Description, MaxDescriptionLength);

        return [new GoalCreated(
            command.UserId, command.IssuedAt, command.CommandId,
            command.GoalId, name, description)];

    static IReadOnlyList<IDomainEvent> Edit(GoalState state, EditGoal command)
    {
        if (state.Archived)

        var name = Text.Required(command.Name, MaxNameLength);
        var description = Text.Optional(command.Description, MaxDescriptionLength);

        return name == state.Name && description == state.Description

            : [new GoalEdited(
                command.UserId, command.IssuedAt, command.CommandId, state.Id, name, description)];
    }

        state.Archived
            ? []
            : [new GoalArchived(command.UserId, command.IssuedAt, command.CommandId, state.Id)];

        state ?? throw new DomainException(ErrorCodes.NotFound, "Goal does not exist.");

        (state, @event) switch

                new GoalState(e.GoalId, e.UserId, e.Name, e.Description, Archived: false),
```

```csharp
using PSPad.Domain.Common;

namespace PSPad.Domain.Tasks;

public static partial class TodoTaskDecider
{
    const int MaxStepTextLength = 200;

    static IReadOnlyList<IDomainEvent> DecideSteps(TodoTaskState? state, ICommand command) =>
        command switch
        {
            AddStep add => Add(Require(state), add),
            EditStep edit => Edit(Require(state), edit),
            CompleteStep complete => CompleteStepOn(Require(state), complete),
            UncompleteStep uncomplete => UncompleteStepOn(Require(state), uncomplete),
            RemoveStep remove => Remove(Require(state), remove),
            ReorderSteps reorder => Reorder(Require(state), reorder),
            _ => []
        };

    static IReadOnlyList<IDomainEvent> Add(TodoTaskState state, AddStep command)
    {
        var text = Text.Required(command.Text, MaxStepTextLength);
        var position = state.Steps.Count;

        return [new StepAdded(
            command.UserId, command.IssuedAt, command.CommandId,
            state.Id, command.StepId, text, command.DueDate, position)];
    }

    static IReadOnlyList<IDomainEvent> Edit(TodoTaskState state, EditStep command)
    {
        var step = RequireStep(state, command.StepId);
        var text = Text.Required(command.Text, MaxStepTextLength);

        return text == step.Text && command.DueDate == step.DueDate
            ? []
            : [new StepEdited(
                command.UserId, command.IssuedAt, command.CommandId,
                state.Id, step.Id, text, command.DueDate)];
    }

    static IReadOnlyList<IDomainEvent> CompleteStepOn(TodoTaskState state, CompleteStep command)
    {
        var step = RequireStep(state, command.StepId);

        return step.CompletedAt is not null
            ? []
            : [new StepCompleted(
                command.UserId, command.IssuedAt, command.CommandId, state.Id, step.Id)];
    }

    static IReadOnlyList<IDomainEvent> UncompleteStepOn(TodoTaskState state, UncompleteStep command)
    {
        var step = RequireStep(state, command.StepId);

        return step.CompletedAt is null
            ? []
            : [new StepUncompleted(
                command.UserId, command.IssuedAt, command.CommandId, state.Id, step.Id)];
    }

    static IReadOnlyList<IDomainEvent> Remove(TodoTaskState state, RemoveStep command)
    {
        var step = RequireStep(state, command.StepId);

        return [new StepRemoved(
            command.UserId, command.IssuedAt, command.CommandId, state.Id, step.Id)];
    }

    static IReadOnlyList<IDomainEvent> Reorder(TodoTaskState state, ReorderSteps command)
    {
        var existing = state.Steps.Select(step => step.Id).ToHashSet();

        if (command.Order.Count != existing.Count || !command.Order.All(existing.Contains))
            throw new DomainException(
                ErrorCodes.StepNotFound, "The new order must list exactly the existing steps.");

        return [new StepsReordered(
            command.UserId, command.IssuedAt, command.CommandId, state.Id, command.Order)];
    }

    static Step RequireStep(TodoTaskState state, StepId stepId) =>
        state.Steps.SingleOrDefault(step => step.Id == stepId)
        ?? throw new DomainException(ErrorCodes.StepNotFound, "Step does not exist.");

    static TodoTaskState? ApplyStep(TodoTaskState? state, IDomainEvent @event) =>
        (state, @event) switch
        {
            ({ } s, StepAdded e) => s with
            {
                Steps = [.. s.Steps, new Step(e.StepId, e.Text, e.DueDate, null, e.Position)]
            },
            ({ } s, StepEdited e) => s with
            {
                Steps = [.. s.Steps.Select(step => step.Id == e.StepId
                    ? step with { Text = e.Text, DueDate = e.DueDate }
                    : step)]
            },
            ({ } s, StepCompleted e) => s with
            {
                Steps = [.. s.Steps.Select(step => step.Id == e.StepId
                    ? step with { CompletedAt = e.OccurredAt }
                    : step)]
            },
            ({ } s, StepUncompleted e) => s with
            {
                Steps = [.. s.Steps.Select(step => step.Id == e.StepId
                    ? step with { CompletedAt = null }
                    : step)]
            },
            ({ } s, StepRemoved e) => s with { Steps = Repack(s.Steps.Where(x => x.Id != e.StepId)) },
            ({ } s, StepsReordered e) => s with
            {
                Steps = [.. e.Order.Select((id, index) =>
                    s.Steps.Single(step => step.Id == id) with { Position = index })]
            },
            _ => state
        };

    /// <summary>Rewrites positions dense from 0, preserving the current order.</summary>
    static IReadOnlyList<Step> Repack(IEnumerable<Step> steps) =>
        [.. steps.OrderBy(step => step.Position)
                 .Select((step, index) => step with { Position = index })];
}
```

```csharp
using PSPad.Domain.Common;
using PSPad.Domain.Tasks;

namespace PSPad.Domain.Inbox;

public static class InboxDecider
{
    const int MaxTextLength = 500;

    public static IReadOnlyList<IDomainEvent> Decide(InboxState state, ICommand command) =>
        command switch
        {
            CaptureToInbox capture => Capture(capture),
            OrganizeInboxItem organize => Organize(state, organize),
            DeleteInboxItem delete => Delete(state, delete),
            _ => []
        };

    static IReadOnlyList<IDomainEvent> Capture(CaptureToInbox command)
    {
        var text = Text.Required(command.Text, MaxTextLength);

        return [new InboxItemCaptured(
            command.UserId, command.IssuedAt, command.CommandId, command.ItemId, text)];
    }

    static IReadOnlyList<IDomainEvent> Organize(InboxState state, OrganizeInboxItem command)
    {
        var item = RequireItem(state, command.ItemId);

        return
        [
            new InboxItemOrganized(
                command.UserId, command.IssuedAt, command.CommandId, item.Id),
            new TaskCreated(
                command.UserId, command.IssuedAt, command.CommandId,
                command.TaskId, command.ListId, item.Text,
                command.DueDate, command.Priority, GoalId: null)
        ];
    }

    static IReadOnlyList<IDomainEvent> Delete(InboxState state, DeleteInboxItem command)
    {
        var item = RequireItem(state, command.ItemId);

        return [new InboxItemDeleted(
            command.UserId, command.IssuedAt, command.CommandId, item.Id)];
    }

    static InboxItem RequireItem(InboxState state, Guid itemId) =>
        state.Items.SingleOrDefault(item => item.Id == itemId)
        ?? throw new DomainException(ErrorCodes.NotFound, "Inbox item does not exist.");

    public static InboxState Apply(InboxState state, IDomainEvent @event) =>
        @event switch
        {
            InboxItemCaptured e => state with
            {
                Items = [.. state.Items, new InboxItem(e.ItemId, e.Text, e.OccurredAt)]
            },
            InboxItemOrganized e => state with
            {
                Items = [.. state.Items.Where(item => item.Id != e.ItemId)]
            },
            InboxItemDeleted e => state with
            {
                Items = [.. state.Items.Where(item => item.Id != e.ItemId)]
            },
            _ => state
        };
}
```
