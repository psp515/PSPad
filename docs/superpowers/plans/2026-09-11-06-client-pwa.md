# Blazor WASM PWA Client Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An installable Blazor WebAssembly PWA with the screens that make PSPad usable day to day: Today, Inbox, areas and lists, task detail with steps, goals, and action history.

**Architecture:** A standalone WASM client talking to the API over HTTP, built with MudBlazor. Every user action becomes a domain command sent to `/api/commands`; every screen reads from a local store that plan 07 turns into a durable IndexedDB replica. This plan keeps that store in memory and online-only, behind the same interface, so plan 07 swaps the implementation without touching a single component.

**Tech Stack:** .NET 10, Blazor WebAssembly, MudBlazor 8.x, bUnit, xUnit, Shouldly

**Spec:** `docs/superpowers/specs/2026-09-11-gtd-core-design.md`

## Global Constraints

- .NET 10 (LTS). All projects target `net10.0`.
- `PSPad.Domain` and `PSPad.Contracts` must compile for WebAssembly — the client references both.
- MudBlazor 8.x. bUnit + xUnit + Shouldly for component tests.
- All user-facing dates are `DateOnly` in the user's time zone.
- The client never reimplements a domain rule: sorting comes from `TaskSort`, Today membership from `TodayRule`.
- Integration tests get their Postgres from Testcontainers via `PostgresFixture` in `PSPad.TestInfrastructure` (AD-9). No compose service is shared with tests.
- Every test class carries `[UnitTest]` or `[IntegrationTest]`; integration classes also join `[Collection(PostgresCollection.Name)]`. Unmarked is a defect.
- No comments in code. Names carry the meaning. The doc comments in this plan's snippets explain things to *you*; port only a comment that states a non-obvious *why*.
- Code, comments, commits and docs in English.

`src/PSPad.Client/State/LocalStore.cs`:

```csharp
using PSPad.Domain.Areas;
using PSPad.Domain.Common;
using PSPad.Domain.Goals;
using PSPad.Domain.Inbox;
using PSPad.Domain.Lists;
using PSPad.Domain.Tasks;

namespace PSPad.Client.State;

/// <summary>
/// In-memory replica of the user's data. Plan 07 backs this with IndexedDB;
/// the interface and the folding logic stay exactly as they are here.
/// </summary>
public class LocalStore : ILocalStore
{
    readonly Dictionary<AreaId, AreaState> _areas = [];
    readonly Dictionary<ListId, TaskListState> _lists = [];
    readonly Dictionary<TaskId, TodoTaskState> _tasks = [];
    readonly Dictionary<GoalId, GoalState> _goals = [];
    InboxState _inbox = new(UserId.Empty, []);

    public IReadOnlyList<AreaState> Areas => [.. _areas.Values.OrderBy(a => a.Order)];
    public IReadOnlyList<TaskListState> Lists => [.. _lists.Values.OrderBy(l => l.Order)];
    public IReadOnlyList<TodoTaskState> Tasks => [.. _tasks.Values];
    public IReadOnlyList<GoalState> Goals => [.. _goals.Values.OrderBy(g => g.Name)];
    public IReadOnlyList<InboxItem> InboxItems =>
        [.. _inbox.Items.OrderByDescending(item => item.CapturedAt)];

    public event Action? Changed;

    public TodoTaskState? Task(TaskId id) => _tasks.GetValueOrDefault(id);

    public virtual void ApplyLocally(IDomainEvent @event)
    {
        switch (@event)
        {
            case AreaCreated or AreaRenamed or AreaArchived:
                Fold(_areas, AreaIdOf(@event), AreaDecider.Apply, @event);
                break;

            case ListCreated or ListRenamed or ListArchived:
                Fold(_lists, ListIdOf(@event), TaskListDecider.Apply, @event);
                break;

            case GoalCreated or GoalEdited or GoalArchived:
                Fold(_goals, GoalIdOf(@event), GoalDecider.Apply, @event);
                break;

            case InboxItemCaptured or InboxItemOrganized or InboxItemDeleted:
                _inbox = InboxDecider.Apply(_inbox with { UserId = @event.UserId }, @event);
                break;

            default:
                if (TaskIdOf(@event) is { } taskId)
                    Fold(_tasks, taskId, TodoTaskDecider.Apply, @event);
                break;
        }

        Changed?.Invoke();
    }

    static void Fold<TKey, TState>(
        Dictionary<TKey, TState> map,
        TKey key,
        Func<TState?, IDomainEvent, TState?> apply,
        IDomainEvent @event)
        where TKey : notnull
        where TState : class
    {
        var next = apply(map.GetValueOrDefault(key), @event);

        if (next is null)
            map.Remove(key);
        else
            map[key] = next;
    }

    static AreaId AreaIdOf(IDomainEvent e) => e switch
    {
        AreaCreated x => x.AreaId,
        AreaRenamed x => x.AreaId,
        AreaArchived x => x.AreaId,
        _ => AreaId.Empty
    };

    static ListId ListIdOf(IDomainEvent e) => e switch
    {
        ListCreated x => x.ListId,
        ListRenamed x => x.ListId,
        ListArchived x => x.ListId,
        _ => ListId.Empty
    };

    static GoalId GoalIdOf(IDomainEvent e) => e switch
    {
        GoalCreated x => x.GoalId,
        GoalEdited x => x.GoalId,
        GoalArchived x => x.GoalId,
        _ => GoalId.Empty
    };

    static TaskId? TaskIdOf(IDomainEvent @event) =>
        @event.GetType().GetProperty(nameof(TaskCreated.TaskId))?.GetValue(@event) as TaskId?;
}
```

`src/PSPad.Client/State/ApiClient.cs`:

```csharp
using System.Net.Http.Json;
using System.Text.Json;
using PSPad.Contracts;
using PSPad.Domain.Common;

namespace PSPad.Client.State;

public sealed record CommandOutcomeDto(
    Guid CommandId, bool Accepted, string? ErrorCode, string? ErrorMessage);

public interface IApiClient
{
    Task<IReadOnlyList<CommandOutcomeDto>> SendAsync(
        IReadOnlyList<ICommand> commands, CancellationToken cancellationToken);
}

public sealed class ApiClient(HttpClient http) : IApiClient
{
    public async Task<IReadOnlyList<CommandOutcomeDto>> SendAsync(
        IReadOnlyList<ICommand> commands, CancellationToken cancellationToken)
    {
        var batch = new CommandBatch(
        [
            .. commands.Select(command => new CommandEnvelope(
                command.GetType().Name,
                JsonSerializer.SerializeToElement(command, command.GetType())))
        ]);

        var response = await http.PostAsJsonAsync("api/commands", batch, cancellationToken);
        response.EnsureSuccessStatusCode();

        var body = await response.Content.ReadFromJsonAsync<CommandBatchResponse>(cancellationToken);

        return
        [
            .. body!.Results.Select(result => new CommandOutcomeDto(
                result.CommandId, result.Accepted, result.ErrorCode, result.ErrorMessage))
        ];
    }
}
```

`src/PSPad.Client/State/CommandDispatcher.cs`:

```csharp
using PSPad.Domain.Areas;
using PSPad.Domain.Common;
using PSPad.Domain.Goals;
using PSPad.Domain.Inbox;
using PSPad.Domain.Lists;
using PSPad.Domain.Tasks;
using PSPad.Domain.Users;

namespace PSPad.Client.State;

public sealed record RejectedCommand(Guid CommandId, string ErrorCode, string Message);

/// <summary>
/// Applies a command locally through the domain, then ships it. Plan 07 inserts
/// an outbox between those two halves; the local half never changes.
/// </summary>
public class CommandDispatcher(LocalStore store, IApiClient api)
{
    readonly List<RejectedCommand> _rejected = [];

    public IReadOnlyList<RejectedCommand> Rejected => _rejected;

    public event Action? RejectionsChanged;

    public virtual async Task SendAsync(
        ICommand command, CancellationToken cancellationToken = default)
    {
        foreach (var @event in Decide(command))
            store.ApplyLocally(@event);

        var outcomes = await api.SendAsync([command], cancellationToken);

        foreach (var outcome in outcomes.Where(o => !o.Accepted))
        {
            _rejected.Add(new RejectedCommand(
                outcome.CommandId,
                outcome.ErrorCode ?? "unknown",
                outcome.ErrorMessage ?? "The server rejected this change."));

            RejectionsChanged?.Invoke();
        }
    }

    public void Dismiss(Guid commandId)
    {
        _rejected.RemoveAll(rejected => rejected.CommandId == commandId);
        RejectionsChanged?.Invoke();
    }

    /// <summary>
    /// Runs the same decide function the server will run. A local failure means
    /// the command was invalid against local state; it is reported and never sent.
    /// </summary>
    IReadOnlyList<IDomainEvent> Decide(ICommand command)
    {
        try
        {
            return command switch
            {
                CreateArea or RenameArea or ArchiveArea =>
                    AreaDecider.Decide(FindArea(command), command),
                CreateList or RenameList or ArchiveList =>
                    TaskListDecider.Decide(FindList(command), command),
                CreateGoal or EditGoal or ArchiveGoal =>
                    GoalDecider.Decide(FindGoal(command), command),
                CaptureToInbox or OrganizeInboxItem or DeleteInboxItem =>
                    InboxDecider.Decide(new InboxState(command.UserId, store.InboxItems), command),
                RegisterUser or SetTimeZone or SetDisplayName => [],
                _ => TodoTaskDecider.Decide(FindTask(command), command)
            };
        }
        catch (DomainException exception)
        {
            _rejected.Add(new RejectedCommand(
                command.CommandId, exception.Code, exception.Message));
            RejectionsChanged?.Invoke();

            return [];
        }
    }

    AreaState? FindArea(ICommand command) => command switch
    {
        CreateArea c => store.Areas.FirstOrDefault(a => a.Id == c.AreaId),
        RenameArea c => store.Areas.FirstOrDefault(a => a.Id == c.AreaId),
        ArchiveArea c => store.Areas.FirstOrDefault(a => a.Id == c.AreaId),
        _ => null
    };

    TaskListState? FindList(ICommand command) => command switch
    {
        CreateList c => store.Lists.FirstOrDefault(l => l.Id == c.ListId),
        RenameList c => store.Lists.FirstOrDefault(l => l.Id == c.ListId),
        ArchiveList c => store.Lists.FirstOrDefault(l => l.Id == c.ListId),
        _ => null
    };

    GoalState? FindGoal(ICommand command) => command switch
    {
        CreateGoal c => store.Goals.FirstOrDefault(g => g.Id == c.GoalId),
        EditGoal c => store.Goals.FirstOrDefault(g => g.Id == c.GoalId),
        ArchiveGoal c => store.Goals.FirstOrDefault(g => g.Id == c.GoalId),
        _ => null
    };

    TodoTaskState? FindTask(ICommand command) =>
        command.GetType().GetProperty(nameof(CreateTask.TaskId))?.GetValue(command)
            is TaskId taskId
            ? store.Task(taskId)
            : null;
}
```

```csharp
using PSPad.Domain.Common;
using PSPad.Domain.Tasks;
using PSPad.Domain.Today;

namespace PSPad.Client.State;

public sealed record TodayRowModel(
    TaskId TaskId,
    string Name,
    string? NextStepText,
    string AreaName,
    Priority Priority,
    bool Starred,
    bool Overdue,
    DateOnly ReasonDate,
    bool Recurring);

public static class TodayProjection
{
    /// <summary>
    /// The Today screen computed from the local replica, using the domain's own
    /// rule and sort — identical to what the server would return (spec §5).
    /// </summary>
    public static IReadOnlyList<TodayRowModel> Rows(ILocalStore store, DateOnly today)
    {
        var lists = store.Lists.ToDictionary(list => list.Id);
        var areas = store.Areas.ToDictionary(area => area.Id);

        var rows = new List<(TodayRowModel Row, TodayVerdict Verdict)>();

        foreach (var task in store.Tasks)
        {
            if (TodayRule.Evaluate(TodayRule.From(task), today) is not { } verdict)
                continue;

            var areaName = lists.TryGetValue(task.ListId, out var list)
                           && areas.TryGetValue(list.AreaId, out var area)
                ? area.Name
                : string.Empty;

            rows.Add((
                new TodayRowModel(
                    task.Id, task.Name, task.NextStep?.Text, areaName,
                    task.Priority, task.Starred, verdict.Overdue, verdict.ReasonDate,
                    task.IsRecurring),
                verdict));
        }

        return
        [
            .. rows
                .OrderByDescending(entry => entry.Verdict.Overdue)
                .ThenBy(entry => TaskSort.Key(
                    entry.Row.Starred, entry.Row.Priority, entry.Verdict.ReasonDate, entry.Row.Name))
                .Select(entry => entry.Row)
        ];
    }
}
```

```csharp
using PSPad.Domain.Common;

namespace PSPad.Client.State;

public sealed record GoalRowModel(GoalId GoalId, string Name, string? Description, int OpenCount, int DoneCount);

public static class GoalProjection
{
    /// <summary>
    /// Goals are global, so their counts deliberately span every area (spec §3.1).
    /// </summary>
    public static IReadOnlyList<GoalRowModel> Rows(ILocalStore store) =>
    [
        .. store.Goals
            .Where(goal => !goal.Archived)
            .Select(goal => new GoalRowModel(
                goal.Id,
                goal.Name,
                goal.Description,
                store.Tasks.Count(task => task.GoalId == goal.Id && !task.IsCompleted),
                store.Tasks.Count(task => task.GoalId == goal.Id && task.IsCompleted)))
            .OrderBy(row => row.Name)
    ];
}
```

```csharp
using System.Net.Http.Json;

namespace PSPad.Client.State;

public sealed record HistoryEntryDto(
    DateTimeOffset OccurredAt, string EventType, string EntityKind, string Summary);

public sealed class HistoryClient(HttpClient http)
{
    public async Task<IReadOnlyList<HistoryEntryDto>> RecentAsync(
        int limit, CancellationToken cancellationToken = default)
    {
        var entries = await http.GetFromJsonAsync<List<HistoryEntryDto>>(
            $"api/history?limit={limit}", cancellationToken);

        return entries ?? [];
    }
}
```
